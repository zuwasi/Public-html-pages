cmake_minimum_required(VERSION 3.20)
project(HeartGCNDemo VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
qt_standard_project_setup()
add_library(heartgcn_core src/model.cpp)
target_include_directories(heartgcn_core PUBLIC include)
add_executable(HeartGCNApp WIN32 src/main.cpp src/main_window.cpp src/visualization_widget.cpp)
target_include_directories(HeartGCNApp PRIVATE src)
target_link_libraries(HeartGCNApp PRIVATE heartgcn_core Qt6::Widgets)
enable_testing()
add_executable(test_model tests/test_model.cpp)
target_link_libraries(test_model PRIVATE heartgcn_core)
add_test(NAME model_reference_checks COMMAND test_model)
